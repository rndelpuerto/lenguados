import {
 OperationBarChart,
 TierComparisonChart,
 DataTable,
 TldrBox,
 NoDataFallback,
 MethodologyNote,
 PartialDataNotice,
 formatOps,
} from '@site/src/components/benchmark';
import { useBenchmarkData } from '@site/src/hooks/use-benchmark-data';

export function MatrixPerformance() {
 const { performance } = useBenchmarkData('math2d');
 if (!performance || !performance.metadata) return <NoDataFallback />;
 const m2 = performance.entities['Matrix2'];
 const m3 = performance.entities['Matrix3'];
 if (!m2 && !m3) return <NoDataFallback />;
 return (
  <>
   <PartialDataNotice summaries={[performance]} />
   <TldrBox>
    {m2 && (
     <>
      <strong>Matrix2: {m2.operations.length} operations</strong>.{' '}
     </>
    )}
    {m3 && (
     <>
      <strong>Matrix3: {m3.operations.length} operations</strong>.{' '}
     </>
    )}
    Matrix operations range from simple component access to O(n³) decompositions.
   </TldrBox>
   {m2 && (
    <>
     <h3>Matrix2 Operations</h3>
     <OperationBarChart data={m2.operations} title="Matrix2 — ops/sec" />
     <MethodologyNote summary="Matrix2 data table">
      <DataTable
       columns={[
        { key: 'name', label: 'Operation' },
        { key: 'opsPerSec', label: 'ops/sec', format: (v) => formatOps(Number(v)), align: 'right' },
        {
         key: 'median',
         label: 'Median (ns)',
         format: (v) => Number(v).toFixed(1),
         align: 'right',
        },
        { key: 'samples', label: 'Samples', align: 'right' },
       ]}
       data={m2.operations}
       highlightBest="opsPerSec"
      />
     </MethodologyNote>
     {m2.tierComparison && m2.tierComparison.operations.length > 0 && (
      <>
       <h4>Matrix2 Tier Comparison</h4>
       <TierComparisonChart data={m2.tierComparison.operations} title="Matrix2 — Tier Comparison" />
      </>
     )}
    </>
   )}
   {m3 && (
    <>
     <h3>Matrix3 Operations</h3>
     <OperationBarChart data={m3.operations} title="Matrix3 — ops/sec" />
     <MethodologyNote summary="Matrix3 data table">
      <DataTable
       columns={[
        { key: 'name', label: 'Operation' },
        { key: 'opsPerSec', label: 'ops/sec', format: (v) => formatOps(Number(v)), align: 'right' },
        {
         key: 'median',
         label: 'Median (ns)',
         format: (v) => Number(v).toFixed(1),
         align: 'right',
        },
        { key: 'samples', label: 'Samples', align: 'right' },
       ]}
       data={m3.operations}
       highlightBest="opsPerSec"
      />
     </MethodologyNote>
     {m3.tierComparison && m3.tierComparison.operations.length > 0 && (
      <>
       <h4>Matrix3 Tier Comparison</h4>
       <TierComparisonChart data={m3.tierComparison.operations} title="Matrix3 — Tier Comparison" />
      </>
     )}
    </>
   )}
  </>
 );
}

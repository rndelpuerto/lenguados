import {
 HeatmapChart,
 BundleSizeChart,
 DataTable,
 TldrBox,
 InsightCallout,
 NoDataFallback,
 MethodologyNote,
 PartialDataNotice,
 formatOps,
 formatKB,
} from '@site/src/components/benchmark';
import { useBenchmarkData } from '@site/src/hooks/use-benchmark-data';

export function Math2dPerformance() {
 const { performance, stress, dx } = useBenchmarkData('math2d');
 if (!performance || !performance.metadata) return <NoDataFallback />;
 const entities = performance.entities;
 const entityNames = Object.keys(entities);
 const heatmapData = [];
 for (const entity of entityNames) {
  for (const op of entities[entity].operations) {
   heatmapData.push({ entity, operation: op.name, opsPerSec: op.opsPerSec });
  }
 }
 const totalOps = entityNames.reduce((sum, e) => sum + entities[e].operations.length, 0);
 const allocation = stress?.allocation ?? [];
 const allZeroAlloc = allocation.length > 0 && allocation.every((a) => a.isZeroAlloc);
 const bundleSize = dx?.bundleSize ?? [];
 const treeShaking = dx?.treeShaking;
 const assertionElimination = dx?.assertionElimination;
 const buildComparison = dx?.buildComparison;
 return (
  <>
   <PartialDataNotice summaries={[performance, stress, dx]} />
   <TldrBox>
    <strong>{totalOps} operations</strong> benchmarked across {entityNames.length} entity types.
    {allZeroAlloc && (
     <>
      {' '}
      All static <code>out</code>-parameter operations verified zero-allocation.
     </>
    )}
    {bundleSize.length > 0 && (
     <>
      {' '}
      Bundle ranges from {(Math.min(...bundleSize.map((b) => b.gzipBytes)) / 1024).toFixed(
       1,
      )} to {(Math.max(...bundleSize.map((b) => b.gzipBytes)) / 1024).toFixed(1)} KB gzip depending
      on import path.
     </>
    )}
   </TldrBox>
   <h3>Throughput Heatmap</h3>
   <p>
    Cell color is mapped on a logarithmic scale because throughput spans several orders of
    magnitude. Empty cells mean the operation does not exist on that entity — hover a cell for exact
    throughput, and see the entity pages below for per-operation detail.
   </p>
   <HeatmapChart data={heatmapData} title="All Entities — ops/sec (log color scale)" />
   <MethodologyNote summary="Heatmap data table">
    <DataTable
     columns={[
      { key: 'entity', label: 'Entity' },
      { key: 'operation', label: 'Operation' },
      { key: 'opsPerSec', label: 'ops/sec', format: (v) => formatOps(Number(v)), align: 'right' },
     ]}
     data={heatmapData}
     highlightBest="opsPerSec"
    />
   </MethodologyNote>
   {bundleSize.length > 0 && (
    <>
     <h3>Bundle Size</h3>
     <p>Each subpath import produces a different bundle. Import only what you need.</p>
     <BundleSizeChart data={bundleSize} title="Gzip Size by Import Path" />
     <MethodologyNote summary="Bundle size data">
      <DataTable
       columns={[
        { key: 'importPath', label: 'Import Path' },
        {
         key: 'rawBytes',
         label: 'Raw',
         format: (v) => formatKB(Number(v)),
         align: 'right',
        },
        {
         key: 'gzipBytes',
         label: 'Gzip',
         format: (v) => formatKB(Number(v)),
         align: 'right',
        },
        {
         key: 'budgetGzipBytes',
         label: 'Budget',
         format: (v) => (v === undefined || v === null ? '—' : formatKB(Number(v))),
         align: 'right',
        },
       ]}
       data={bundleSize}
      />
     </MethodologyNote>
    </>
   )}
   {treeShaking && (
    <>
     <h3>Tree-Shaking</h3>
     <DataTable
      columns={[
       { key: 'metric', label: 'Metric' },
       { key: 'value', label: 'Value', align: 'right' },
      ]}
      data={[
       { metric: 'Full library', value: formatKB(treeShaking.fullSize) },
       { metric: 'Vector2 only', value: formatKB(treeShaking.minimalSize) },
       { metric: 'Reduction', value: `${treeShaking.reductionPercent.toFixed(0)}%` },
       {
        metric: 'sideEffects: false',
        value: treeShaking.sideEffectsEmpty ? 'Honored' : 'Not honored',
       },
      ]}
     />
    </>
   )}
   {assertionElimination && (
    <>
     <h3>Assertion Elimination</h3>
     <DataTable
      columns={[
       { key: 'check', label: 'Check' },
       { key: 'status', label: 'Status' },
      ]}
      data={[
       {
        check: 'Assertion names in production',
        status: assertionElimination.prodBundle.containsAssertions
         ? 'Export stubs present'
         : 'None',
       },
       {
        check: 'DEV_MODE in production',
        status: assertionElimination.prodBundle.containsDevMode
         ? 'FOUND (issue)'
         : 'None (correct)',
       },
       {
        check: 'NODE_ENV guard in production',
        status: assertionElimination.prodBundle.containsNodeEnv
         ? 'Present (runtime DCE)'
         : 'None (static DCE)',
       },
       {
        check: 'Assertions in development',
        status: assertionElimination.devBundle.containsAssertions
         ? 'Present (correct)'
         : 'Missing (issue)',
       },
      ]}
     />
    </>
   )}
   {buildComparison && (
    <>
     <h3>Build Size Comparison</h3>
     <DataTable
      columns={[
       { key: 'build', label: 'Build' },
       { key: 'raw', label: 'Raw', align: 'right' },
       { key: 'gzip', label: 'Gzip', align: 'right' },
      ]}
      data={[
       {
        build: 'Development',
        raw: formatKB(buildComparison.devSize.raw),
        gzip: formatKB(buildComparison.devSize.gzip),
       },
       {
        build: 'Production',
        raw: formatKB(buildComparison.prodSize.raw),
        gzip: formatKB(buildComparison.prodSize.gzip),
       },
       {
        build: 'Reduction',
        raw: `${buildComparison.reductionPercent.raw.toFixed(1)}%`,
        gzip: `${buildComparison.reductionPercent.gzip.toFixed(1)}%`,
       },
      ]}
     />
    </>
   )}
   {allocation.length > 0 && (
    <>
     <h3>Zero-Allocation Verification</h3>
     <MethodologyNote summary="Allocation data table">
      <DataTable
       columns={[
        { key: 'entity', label: 'Entity' },
        { key: 'operation', label: 'Operation' },
        { key: 'variant', label: 'Variant' },
        { key: 'bytesPerOp', label: 'B/op', format: (v) => Number(v).toFixed(1), align: 'right' },
        { key: 'isZeroAlloc', label: 'Zero-Alloc', format: (v) => (v ? 'YES' : 'NO') },
       ]}
       data={allocation}
      />
     </MethodologyNote>
    </>
   )}
  </>
 );
}
